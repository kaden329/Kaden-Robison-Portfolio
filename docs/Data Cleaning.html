    pressure = `depSM: Depth [salt water, m]`,

Joining Metadata

ctd_meta <- ctd_subset |> 
  left_join(metadata, by = "station") |> 
  select(all_of(names(metadata)), everything())

ctd_meta <- ctd_meta |> 
  drop_na(station, salinity, temperature, pressure)

ctd_meta <- ctd_meta |>  
  filter(station != "40")

ctd_meta <- ctd_meta %>%
  rename(depth = `ml_depth_[m]`)

Averaging Oxygen

ctd_avg <- ctd_meta %>%
  mutate(depth_bin = round(depth / 10) * 10) %>%  # 10 m bins   
  group_by(depth_bin) %>%
  summarise(oxygen = mean(oxygen, na.rm = TRUE)) %>%
  ungroup()

First Visualizations

ggplot(ctd_avg, aes(x = oxygen, y = depth_bin)) +
  geom_path(color = "blue", linewidth = 1.2) +   # main line
